Angle brackets are substituted back after rendering, so the markup survives with unsafe left off; the body is rendered here with markdownify instead, and still takes links, code spans and lists. */ -}} {{- $kind := .Get "kind" | default "note" -}} {{- $labels := dict "note" "NOTE" "warn" "WATCH OUT" -}} {{- $label := index $labels $kind -}} {{- if not $label }}{{ errorf "callout: unknown kind %q in %s — use note or warn" $kind .Position }}{{ end -}}
